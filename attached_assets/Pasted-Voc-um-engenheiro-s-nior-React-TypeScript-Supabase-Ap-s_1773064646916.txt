CAUSA RAIZ — loading nunca vira false
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

O setLoading(false) não está sendo chamado em algum
caminho de erro ou exception no AuthContext.

Cenários que travam o loading no mobile:

1. getSession() lança erro não capturado → setLoading(false) 
   nunca é chamado

2. refreshProfile() lança erro → código para antes 
   do setLoading(false)

3. onAuthStateChange nunca dispara no mobile browser
   (comum em Safari/iOS com third-party cookies bloqueados)

4. getSession() e onAuthStateChange disputam o setLoading,
   causando estado inconsistente

Substitua TODO o useEffect de autenticação por esta versão
defensiva e à prova de qualquer caminho de erro:

  useEffect(() => {
    let mounted = true

    const initAuth = async () => {
      try {
        const { data: { session }, error } = await supabase.auth.getSession()
        
        if (!mounted) return
        
        if (error) {
          console.error('getSession error:', error)
          setSession(null)
          setUser(null)
          setLoading(false)
          return
        }

        setSession(session)
        setUser(session?.user ?? null)

        if (session?.user) {
          try {
            await refreshProfile(session.user.id)
          } catch (profileError) {
            console.error('refreshProfile error:', profileError)
            // NÃO bloquear o loading por erro de profile
          }
        }
      } catch (err) {
        console.error('initAuth error:', err)
      } finally {
        // SEMPRE liberar o loading, sem exceção
        if (mounted) setLoading(false)
      }
    }

    initAuth()

    const { data: { subscription } } = supabase.auth.onAuthStateChange(
      async (event, session) => {
        if (!mounted) return

        setSession(session)
        setUser(session?.user ?? null)

        if (session?.user) {
          try {
            await refreshProfile(session.user.id)
          } catch (profileError) {
            console.error('profile error on auth change:', profileError)
          }
        }

        // Garantir que loading seja false após qualquer evento
        if (mounted) setLoading(false)
      }
    )

    // Timeout de segurança — se após 5s ainda loading, liberar
    const safetyTimeout = setTimeout(() => {
      if (mounted) {
        console.warn('Auth timeout — forcing loading false')
        setLoading(false)
      }
    }, 5000)

    return () => {
      mounted = false
      subscription.unsubscribe()
      clearTimeout(safetyTimeout)
    }
  }, [])
